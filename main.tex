\section{Introduction to radiative transfer}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{\textcolor{activecolor}{Introduction}}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
        \item The Sun heats the Earth (short wavelengths); the Earth re-emits in the \textbf{infrared}, which the air partly absorbs and re-emits: this is the \textbf{greenhouse effect}.
        \item \textbf{Goal}: compute the atmospheric temperature $\textcolor{pythonblue}{\temperature (\bol{x})}$ in \textbf{3D}%, accounting for terrain (Chamonix valley), clouds and CO$_2$.
        \item \textbf{Difficulty}: the radiative intensity $\textcolor{darkgreen}{\intensity (\bol{x},\bol{\omega})}$ depends on position (3), direction (2) and frequency (1). The problem has \textbf{7 dimensions}.
        \item \textbf{Motivation}: assess the impact of aviation, where CO$_2$ accounts for only about 1/3 of the warming (the rest comes from NO$_x$ and contrails).
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \begin{block}{Planck's law (black body)}
        \[ B_\nu(\temperature)=\frac{\nu^3}{e^{\nu/\temperature}-1} \]
        {\small Sun: $T_S\approx 5800\,$K\\ Earth: $T_E\approx 288\,$K}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{\textcolor{activecolor}{Radiative transfer equations I}}

  \begin{block}{Radiative transfer along a ray}
    \[
      \bol{\omega}\cdot\nabla \intensity + \kappa_\nu \intensity = \kappa_\nu B_\nu(\temperature),
      \qquad \jintensity := \frac{1}{4\pi}\int_{\mathbb{S}^2} \intensity(\bol{\omega})\dd\bol{\omega}
    \]
  \end{block}
  \medskip 
  \begin{block}{Energy balance of the air}
    \[
      \rho\,\mathbf{u}\cdot\nabla \temperature - \kappa_T\Delta \temperature
      + c_2\!\int_0^\infty\!\kappa_\nu B_\nu(\temperature)\dd\nu
      = c_2\!\int_0^\infty\!\kappa_\nu \jintensity\dd\nu
    \]
  \end{block}
  \medskip
  \begin{itemize}[label=$\square$]\setlength\itemsep{0.8em}
    \item $\kappa_\nu(\bol{x})$: \textbf{absorption} coefficient (air, CO$_2$, clouds); $\rho$: air density; $\mathbf{u}$: wind (Navier--Stokes).
    \item The coupling: the air \textbf{emits} $\kappa_\nu B_\nu(\temperature)$ and \textbf{absorbs} $\kappa_\nu \jintensity$; the difference heats or cools it.
    %\item The general case adds a scattering albedo $a_\nu$ (clouds) and reflective boundary conditions.
  \end{itemize}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{\textcolor{activecolor}{Radiative transfer equations II -- eliminating the directions}}
  
  Along a segment, the equation can be solved explicitly (Beer--Lambert law):
  \[
    I(\bol{x},\bol{\omega}) = I(\bol{x}_\Gamma)\,e^{-\int_{[\bol{x},\bol{x}_\Gamma]}\kappa}
    + \int_{[\bol{x},\bol{x}_\Gamma]} e^{-\int_{[\bol{x},\bol{x}']}\kappa}\,S(\bol{x}')\dd\bol{x}'
  \]
  Averaging over $\mathbb{S}^2$, $J$ depends only on $\temperature$ and on the boundary:
  \begin{align*}
    \jintensity[\temperature](\bol{x}) &= \underbrace{\frac{1}{4\pi}\int_\Omega \kappa_\nu B_\nu\big(\temperature(\bol{x}')\big)\frac{e^{-\int_{[\bol{x},\bol{x}']}\kappa_\nu}}{|\bol{x}'-\bol{x}|^2}\dd\bol{x}'}_{\text{Volume integral}}\\
    &\quad+ \underbrace{\frac{1}{4\pi}\int_\Gamma \intensity\Big(\bol{x},\tfrac{\bol{x}'-\bol{x}}{|\bol{x}'-\bol{x}|}\Big)\frac{[(\bol{x}'-\bol{x})\cdot\bol{n}(\bol{x}')]^-}{|\bol{x}'-\bol{x}|^3}e^{-\int_{[\bol{x},\bol{x}']}\kappa_\nu}\dd\Gamma(\bol{x}')}_{\text{Surface integral}}
  \end{align*}
  \alert{ $\Longrightarrow$ We go from 7 to 4 dimensions: $(\bol{x},\nu)$.}
\end{frame}
\section{Numerical scheme}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{\textcolor{activecolor}{Numerical scheme \small {( PH. Tournier $\&$ O. Pironneau) } }}
  \begin{block}{Fixed-point iteration (convergence and monotonicity proven in \texttt{F. Golse \& O. Pironneau [JCP-2022]})}
    \[
      \mathbf{u}\cdot\nabla \temperature^{n+1}-\kappa_T\Delta \temperature^{n+1}+c_2\!\int_0^\infty\!\kappa_\nu B_\nu(\temperature^{n+1})\dd\nu
      = c_2\!\int_0^\infty\!\kappa_\nu J[\temperature^{n}]_\nu\dd\nu
    \]
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{At each iteration $n$}:
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
        \item compute $J[\temperature^n]_\nu$: volume integral $+$ surface integral;
      \item solve the nonlinear temperature equation (\textbf{Newton}) to get $\temperature^{n+1}$.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Discretization}: $P^1$ finite elements on a tetrahedral mesh ($N$ vertices).
      \[ \mathbf{j}_\nu = \mathbf{G}_\kappa\,\mathbf{b}_\nu,\quad \mathbf{G}_\kappa\in\mathbb{R}^{N\times N} \]
      \alert{$\mathbf{G}_\kappa$ is dense: $O(N^2)$ memory.}
    \end{column}
  \end{columns}
  %\medskip
  %\textbf{Frequency integral}: the kernels depend on $\nu$ only through $\kappa_\nu$, so frequencies are grouped by value of $\kappa$ (8 values instead of 683 measurements).
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{\textcolor{activecolor}{Hierarchical matrices in a nutshell}}
  \begin{columns}[T]
    \begin{column}{0.60\textwidth}
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.9em}
        %\item The kernel $e^{-\int\kappa}/|\bol{x}'-\bol{x}|^2$ decays quickly: two \textbf{distant} groups of vertices interact in a simple way.
        \item The kernel $\frac{e^{-\int_{[\bol{x},\bol{x}']}\kappa_\nu}}{|\bol{x}'-\bol{x}|^2}$ decays quickly: two \textbf{distant} groups of vertices interact in a simple way.
        \item Their block is \textbf{low-rank}: $\mathbf{B}\approx\sum_{j=1}^{r}\sigma_j\mathbf{u}_j\mathbf{v}_j^{T}$, $r\ll N$.
        \item The SVD costs $O(N^3)$: we use \textbf{ACA} (Adaptive Cross Approximation), which needs only $\sim 2rN$ coefficients.
        \item A geometric \textbf{cluster tree} partitions the matrix; a block is compressed if $\max(\operatorname{diam}X,\operatorname{diam}Y)\le\eta\operatorname{dist}(X,Y)$.
        \item \textbf{Gain}: memory and matrix-vector product go from $O(N^2)$ to almost linear.
        \item Implementation: \texttt{FreeFEM} + \texttt{Htool} library \small(by P. Marchand \& P-H. Tournier)  .
      \end{itemize}
    \end{column}
    \begin{column}{0.36\textwidth}

\begin{frame}{\textcolor{activecolor}{Speeding up with ray tracing}}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \medskip
      The radiation reaching $\bol{x}$ travels along a segment inside the air $\Omega$. If a mountain blocks $[\bol{x},\bol{x}']$, no light goes through it, and the contribution of $\bol{x}'$ is zero (the ground is accounted for by the surface term). So the kernel carries a visibility factor:
      \[
        G^{ij}_\kappa=\frac{1}{4\pi}\int_\Omega
        \underbrace{V(\bol{x}^i,\bol{x}')}_{\in\{0,1\}}\,\kappa\,
        \frac{e^{-\int_{[\bol{x}^i,\bol{x}']}\kappa}}{|\bol{x}^i-\bol{x}'|^2}\,\hat w^j(\bol{x}')\dd\bol{x}'
      \]
      \vspace{-0.4em}
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
        \item \textbf{Skip} the line integral $\int_{[\bol{x},\bol{x}']}\kappa$ for every hidden pair: many matrix entries are exactly $0$.
        \item \textbf{Early stop}: while tracing a ray, stop once the optical depth $\int\kappa$ is large, since $e^{-\int\kappa}\approx 0$.
        \item $V$ does not depend on $\nu$: compute it \textbf{once} and reuse it for every value of $\kappa$.
      \end{itemize}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering

\section{Raytracing}
\begin{frame}{\textcolor{activecolor}{What is a ray?}}
 \begin{columns}[T]
    % Left: definition
    \begin{column}{0.42\textwidth}
      %\small
      \vspace{1.2cm}
      A \textbf{ray} is a half-line defined by two quantities:
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
        \item an \textbf{origin} $\mathbf{o}$ (where it starts),
        \item a \textbf{direction} $\mathbf{d}$ (where it goes, usually a unit vector).
      \end{itemize}
 
      \medskip
      Every point on the ray is given by the parameter $t$:
      \[
        \textcolor{blue!70!black}{P(t)} = \mathbf{o} + t\,\textcolor{red!70!black}{\mathbf{d}}
      \]
      for $t \in [t_{\min}, t_{\max}]$.  
        
      \medskip
      \visible<2->{ 
      \alert{Core question:} what is the first surface hit?
    }
    \end{column}

\begin{frame}{\textcolor{activecolor}{The quadratic wall}}
  \medskip
  \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}

    \item Each pair = one \alert{ray}, each ray must be tested against the obstacles.

\begin{frame}{\textcolor{activecolor}{BVH: a hierarchy of bounding boxes}}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \vspace{1cm}
        \begin{center}
  \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
    \item Each node = an axis-aligned box (AABB) containing its whole subtree.
    \item The \alert{leaves} hold a few triangles.
    \item Building the BVH is expensive, the cost is amortized over millions of rays
    \item If a ray misses a box, it misses \alert{everything inside it}.
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
      \item Test the ray against the root box (ray-box test, very cheap).
      \item If hit: descend into both children; otherwise \alert{prune} the whole subtree.
      \item On reaching a leaf: test the ray against its point.
      \item Stop at the first hit.
      \end{itemize}
 \end{itemize}
  \end{center}
    \end{column}

\begin{frame}{\textcolor{activecolor}{The hardware side: RT Cores}}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \vspace{1.8cm}
      Modern NVIDIA GPUs (RTX) contain a \alert{dedicated circuit} for ray tracing, the \alert{RT Core}:
      \begin{itemize}[label=$\square$]\setlength\itemsep{0.8em}
        \item The regular GPU cores are \emph{general-purpose}: they run any program, in parallel.
        \item The RT Core does \alert{one thing only}: given a ray, it finds the first object hit, by descending the tree of nested boxes (the BVH).
        \item It is the same operation repeated billions of times: ideal to wire directly into silicon.
        \item The general-purpose cores just \alert{send} the ray and \alert{read} the answer; meanwhile they stay free for other computations.
      \end{itemize}
    \end{column}

    \node[box=blue,text width=3.2cm,right=of c]   (d) {\textbf{4. Collect}\\[1pt]\scriptsize bits $\rightarrow$ matrix $V$};
    \draw[arr] (a)--(b);
    \draw[arr] (b)--(c);
    \draw[arr] (c)--(d);

  \end{center}
  \vspace{3mm}
  \begin{itemize}[label=$\square$]\setlength\itemsep{0.6em}
    \item \alert{Input}: points $x_1,\dots,x_N$ and the obstacle mesh.
    \item \alert{Output}: the matrix $V\in\{0,1\}^{N\times N}$, with $V_{ij}=1$ iff the segment $[x_i,x_j]$ meets no obstacle.
    \item \alert{Cost}: $\displaystyle \frac{N(N-1)}{2}$ independent tests; only step 3 scales with $N^2$, and it runs on the RT Cores.
    \item Called from \texttt{FreeFEM++} as \texttt{OptixBuildVisibilityPairs3D(Th3, ThS, V)}.
    \item The \texttt{.cu} is compiled separately to .ptx (\texttt{nvcc -ptx}) and loaded at runtime
  \end{itemize}

\begin{frame}[fragile]{\textcolor{activecolor}{The raygen program}}
\begin{center}
  
  \begin{lstlisting}
extern "C" __global__ void __raygen__rg() {
  const unsigned long long k = optixGetLaunchIndex().x;
  if (k >= params.numPairs) return;
  unsigned a, b;  unrank(k, params.N, a, b);       // k -> (i,j) global idx

  float3 P = params.pts[a], Q = params.pts[b];
  float3 d = Q - P;  float L = sqrtf(dot3(d,d));   // segment length
  d /= L;                                          // unit direction

  unsigned occluded = 1;                           // pessimistic
  optixTrace(params.gas, P, d,
             params.tmin, params.tmax, 0.0f,   // tmin, tmax, time
             OptixVisibilityMask(255),
             TERMINATE_ON_FIRST_HIT | DISABLE_CLOSESTHIT,
             0, 1, 0, occluded);
  if (!occluded) atomicOr(&params.bits[k >> 5], 1u << (k & 31));
}
\end{lstlisting}
\end{center}
 \begin{itemize}
        \item \alert{Interval} $[t_{\min},t_{\max}]=[10^{-6}L,\ (1-10^{-5})L]$: the ray starts \alert{just after} $P$ and stops \alert{just before} $Q$.
        \item \texttt{TERMINATE\_ON\_FIRST\_HIT}: a single obstacle is enough to conclude ``blocked''.
        \item \texttt{DISABLE\_CLOSESTHIT}: we do not compute the closest hit (useless here).
        \item \alert{Payload} = 1 integer (\texttt{occluded}) shared between raygen and miss.
        \item Mask 255: the ray sees all the geometry.
  \end{itemize}
\end{frame}
